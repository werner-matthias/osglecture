% langselect's target language is bridged from OLLM's own per-language
% build value (\OsgLectureLanguage is not yet defined this early in class
% loading, but the build-spec accessor it is built from already is), so a
% single langselect-aware source produces the correct language without any
% further per-target configuration.
\OsgLectureBuildLoadedTF
  { \edef\olsTargetLanguage{\OsgLectureBuildValue{language}} }
  { }
% osglecture.cls defers any \usepackage/\RequirePackage written directly
% in projectconfig.tex until after the target's base class (here
% ltx-talk) has loaded -- packages that themselves probe "does the base
% class already define X", as langselect's default 'unified shorthands'
% does via csquotes, would otherwise see a false negative this early and
% could collide with the base class's own later definition. \title below
% is deferred the same way (osglecture's existing metadata-field queue),
% replayed in the same order right after \LoadClass, so \lende is already
% available by the time it runs -- no special handling needed here.
\usepackage[languages={en,de}]{langselect}
\title{\lende{One source, multiple documents}{Eine Quelle, mehrere Dokumente}}
\author<presentation>[M.~Mustermann]{Max Mustermann}
\author<longform>[Mustermann]{Max Mustermann}
\LectureTargetSetup{talk}{profile=series-ltx-talk}
\OsgLectureSetup{
  mode/longform = {
    continuation = {counters={page,section}}
  }
}
